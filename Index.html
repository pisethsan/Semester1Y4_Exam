function dataCalculation() {
  var sum = 0;
  var obj = document.getElementsByName("myChoice");
  for (var i = 0; i < obj.length; i++) {
    if (!obj[i].checked) continue;
    sum += parseFloat(obj[i].value);
  }
  alert("Total Price is: " + sum + "$");
}
</script>

១. ដូចម្តេចដែលហៅថាការសរសេរកម្មវិធីបែប OOP?

OOP (Object-Oriented Programming) គឺជារបៀបសរសេរកម្មវិធីដែលផ្អែកលើ Class និង Object ដើម្បីរៀបចំទិន្នន័យ និងមុខងារឱ្យមានរបៀបរៀបរយ ងាយស្រួលប្រើប្រាស់ និងកែប្រែ[cite: 16]។

គោលគំនិតសំខាន់ទាំង ៤៖

  • Encapsulation: ការការពារទិន្នន័យដោយប្រើ Access Modifiers (private, protected, public)[cite: 16]។
  • Inheritance: អនុញ្ញាតឱ្យ Subclass ទទួលយក attributes និង methods ពី Superclass[cite: 16]។
  • Polymorphism: Method អាចមានឈ្មោះដូចគ្នា ប៉ុន្តែមានដំណើរការ/អាកប្បកិរិយាខុសគ្នា[cite: 16]។
  • Abstraction: ការលាក់បាំងភាពស្មុគស្មាញខាងក្នុង និងបង្ហាញតែមុខងារសំខាន់ៗ[cite: 16]។

២. ដូចម្តេចដែលហៅថា Constructor? ហើយ Constructor មានប៉ុន្មានអ្វីខ្លះ?

Constructor គឺជា Method ពិសេសមួយដែលត្រូវបានហៅដោយស្វ័យប្រវត្តិនៅពេលបង្កើត Object ថ្មី ដើម្បីកំណត់តម្លៃដំបូងឱ្យ Attribute[cite: 16]។

Constructor ចែកចេញជា ៣ ប្រភេទសំខាន់ៗ៖

  • Default Constructor: Constructor ដែលមិនមាន parameter សម្រាប់កំណត់តម្លៃដើមស្រាប់[cite: 16]។
  • Parameterized Constructor: Constructor ដែលមាន parameter ដើម្បីអនុញ្ញាតឱ្យយើងបញ្ជូនតម្លៃខុសៗគ្នាទៅឱ្យ instance variables ពេល instantiate object ថ្មី[cite: 16]។
  • Copy Constructor: Constructor ដែលទទួល object នៃ class នោះជា parameter ដើម្បីចម្លងទិន្នន័យពី object មួយទៅ object ថ្មីមួយទៀត[cite: 16]។

Class គឺជាពុម្ពគំរូ (Blueprint ឬ Template) សម្រាប់បង្កើត Objects ដែលកំណត់នូវលក្ខណៈសម្បត្តិ (Attributes/Fields) និងសកម្មភាព (Methods) ដែល Object នៃ Class នោះត្រូវមាន[cite: 16]។

របៀបនៃការបង្កើត Class: ការបង្កើត Class ក្នុង Java គឺប្រើប្រាស់ keyword class តាមដោយឈ្មោះ Class រួចកំណត់នូវ attributes, constructor, និង methods នៅខាងក្នុង curly braces {}[cite: 16]។

import java.util.ArrayList;
import java.util.List;

// 1. Class Person
class Person {
    protected String name;
    protected int age;
    protected String gender;

    public Person(String name, int age, String gender) {
        this.name = name;
        this.age = age;
        this.gender = gender;
    }

    public void info() {
        System.out.println("Name: " + name + ", Age: " + age + ", Gender: " + gender);
    }
}

// 2. Employee inheritance from Person
class Employee extends Person {
    protected String employee_id;
    protected String position;
    protected double salary;

    public Employee(String name, int age, String gender, String employee_id, String position, double salary) {
        super(name, age, gender);
        this.employee_id = employee_id;
        this.position = position;
        this.salary = salary;
    }

    @Override
    public void info() {
        super.info();
        System.out.println("Employee ID: " + employee_id + ", Position: " + position + ", Salary: $" + String.format("%.2f", salary));
    }
}

// 3. Student inheritance from Person
class Student extends Person {
    private String student_id;
    private String major;
    private List<Double> scores;

    public Student(String name, int age, String gender, String student_id, String major) {
        super(name, age, gender);
        this.student_id = student_id;
        this.major = major;
        this.scores = new ArrayList<>();
    }

    public void add_score(double score) {
        this.scores.add(score);
    }

    @Override
    public void info() {
        super.info();
        System.out.println("Student ID: " + student_id + ", Major: " + major + ", Scores: " + scores);
    }
}

// 4. Lecturer inheritance from Employee
class Lecturer extends Employee {
    protected String lecturer_id;
    protected String department;
    protected List<String> subjects;

    public Lecturer(String name, int age, String gender, String employee_id, String position, double salary,
                    String lecturer_id, String department) {
        super(name, age, gender, employee_id, position, salary);
        this.lecturer_id = lecturer_id;
        this.department = department;
        this.subjects = new ArrayList<>();
    }

    public void add_subject(String subject_name) {
        this.subjects.add(subject_name);
    }

    @Override
    public void info() {
        super.info();
        System.out.println("Lecturer ID: " + lecturer_id + ", Department: " + department + ", Subjects: " + subjects);
    }
}

// 5. Subject inheritance from Lecturer
class Subject extends Lecturer {
    private String subject_id;
    private String subject_name;
    private int credit;

    public Subject(String name, int age, String gender, String employee_id, String position, double salary,
                   String lecturer_id, String department, String subject_id, String subject_name, int credit) {
        super(name, age, gender, employee_id, position, salary, lecturer_id, department);
        this.subject_id = subject_id;
        this.subject_name = subject_name;
        this.credit = credit;
    }

    @Override
    public void info() {
        super.info();
        System.out.println("Subject ID: " + subject_id + ", Subject Name: " + subject_name + ", Credit: " + credit);
    }
}
